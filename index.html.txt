        function copyText(elementId, feedbackId) {
            const text = document.getElementById(elementId).textContent;
            const tempInput = document.createElement('input');
            tempInput.value = text;

        // Golden Fireflies & Twinkling Stars Particle Effect
        const fireflyCanvas = document.getElementById('firefly-canvas');
        const fCtx = fireflyCanvas.getContext('2d');
        let magicSparkles = [];

        function resizeFireflyCanvas() {
            fireflyCanvas.width = window.innerWidth;
            fireflyCanvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeFireflyCanvas);
        resizeFireflyCanvas();

        class MagicSparkle {
            constructor() {
                this.reset(true);
            }
            reset(initial = false) {
                this.x = Math.random() * fireflyCanvas.width;
                this.y = initial
                    ? Math.random() * fireflyCanvas.height
                    : fireflyCanvas.height + 10;

                // Made even smaller and more delicate
                this.size = Math.random() * 0.75 + 0.25;
                // Gentle levitating luxury drift
                this.speedY = -(Math.random() * 0.2 + 0.05);
                this.swayAmplitude = Math.random() * 0.2 + 0.05;
                this.swaySpeed = Math.random() * 0.015 + 0.005;
                this.phase = Math.random() * Math.PI * 2;

                // Twinkle / chikchik cycle (sharp diamond glint)
                this.twinkleSpeed = Math.random() * 0.09 + 0.05;
                this.twinklePhase = Math.random() * Math.PI * 2;
                this.isStarCross = Math.random() < 0.6; // 60% sharp 4-point diamond glints
            }
            update() {
                this.phase += this.swaySpeed;
                this.twinklePhase += this.twinkleSpeed;

                this.x += Math.sin(this.phase) * this.swayAmplitude;
                this.y += this.speedY;

                if (this.y < -15 || this.x < -15 || this.x > fireflyCanvas.width + 15) {
                    this.reset(false);
                }
            }
            draw() {
                const rawWave = (Math.sin(this.twinklePhase) + 1) * 0.5;
                const alpha = Math.pow(rawWave, 4.5);

                if (alpha < 0.05) return;

                fCtx.save();
                fCtx.translate(this.x, this.y);

                // Soft luxury gold aura
                const glowRadius = this.size * 3.8;
                const glow = fCtx.createRadialGradient(0, 0, 0, 0, 0, glowRadius);
                glow.addColorStop(0, `rgba(255, 253, 235, ${alpha})`);
                glow.addColorStop(0.35, `rgba(232, 197, 101, ${alpha * 0.6})`);
                glow.addColorStop(1, 'rgba(212, 175, 55, 0)');
                fCtx.fillStyle = glow;
                fCtx.beginPath();
                fCtx.arc(0, 0, glowRadius, 0, Math.PI * 2);
                fCtx.fill();

                // 4-point diamond sparkle flare ("chikchik" magic glint) scaled smaller
                if (this.isStarCross && alpha > 0.25) {
                    const rayLen = this.size * (3.0 + rawWave * 3.5);
                    fCtx.strokeStyle = `rgba(255, 255, 255, ${alpha})`;
                    fCtx.lineWidth = 0.5;
                    fCtx.beginPath();
                    fCtx.moveTo(-rayLen, 0);
                    fCtx.lineTo(rayLen, 0);
                    fCtx.moveTo(0, -rayLen);
                    fCtx.lineTo(0, rayLen);
                    fCtx.stroke();
                }

                // Crisp bright core
                fCtx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
                fCtx.beginPath();
                fCtx.arc(0, 0, this.size * 0.55, 0, Math.PI * 2);
                fCtx.fill();

                fCtx.restore();
            }
        }

        const particleCount = window.innerWidth < 640 ? 28 : 50;
        for (let i = 0; i < particleCount; i++) {
            magicSparkles.push(new MagicSparkle());
        }

        function animateFireflies() {
            fCtx.clearRect(0, 0, fireflyCanvas.width, fireflyCanvas.height);
            for (let i = 0; i < magicSparkles.length; i++) {
                magicSparkles[i].update();
                magicSparkles[i].draw();
            }
            requestAnimationFrame(animateFireflies);
        }

        window.addEventListener('load', () => {
            animateFireflies();
        });
        if (document.readyState === 'complete') {
            animateFireflies();
        }

        // Enhanced Dynamic Scroll Smoke & Speed Effect
        let lastScrollTop = window.pageYOffset || document.documentElement.scrollTop;
        let scrollVelocity = 0;
        const scrollSmokeOverlay = document.getElementById('scroll-smoke-overlay');
        let scrollTimeout;

        window.addEventListener('scroll', () => {
            const st = window.pageYOffset || document.documentElement.scrollTop;
            const delta = Math.abs(st - lastScrollTop);
            lastScrollTop = st <= 0 ? 0 : st;

            // Compute vibrant velocity response for golden smoke
            scrollVelocity = Math.min(delta * 0.12, 1);
            
            // Boost golden glow opacity based on scroll speed
            const activeOpacity = 0.15 + (scrollVelocity * 0.65);
            scrollSmokeOverlay.style.opacity = activeOpacity;

            // Dynamically boost firefly / sparkle speed when scrolling faster
            for (let i = 0; i < magicSparkles.length; i++) {
                magicSparkles[i].speedY = -(Math.random() * 0.3 + 0.1 + (scrollVelocity * 0.8));
            }

            clearTimeout(scrollTimeout);
            // When scroll stops, smoothly fade back to base golden glow state
            scrollTimeout = setTimeout(() => {
                scrollSmokeOverlay.style.opacity = '0.15';
                for (let i = 0; i < magicSparkles.length; i++) {
                    magicSparkles[i].speedY = -(Math.random() * 0.2 + 0.05);
                }
            }, 180);
        }, { passive: true });
